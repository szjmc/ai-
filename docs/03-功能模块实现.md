### client-api：用 XHR 换进度条

`fetch` 无法上报上传进度，这里是全项目唯一用 `XMLHttpRequest` 的地方：

```ts
// src/lib/client-api.ts
export function uploadFile(form: FormData, onProgress: (value: number) => void): Promise<ApiBody<FileDict>> {
  return new Promise((resolve, reject) => {
    const xhr = new XMLHttpRequest();
    xhr.open("POST", "/api/v1/file/uploadFile");
    xhr.timeout = 150000;
    xhr.upload.onprogress = (event) => {
      if (event.lengthComputable) onProgress(Math.round(event.loaded / event.total * 100));
    };
    xhr.onerror = () => reject(new Error("网络连接中断，请刷新列表确认上传结果"));
    xhr.ontimeout = () => reject(new Error("上传超时，请刷新列表确认是否已入库，避免重复上传"));
    xhr.onload = () => {
      try {
        const body = JSON.parse(xhr.responseText) as ApiBody<FileDict>;
        if (xhr.status >= 200 && xhr.status < 300 && body.status < 400 && body.data?.id) resolve(body);
        else reject(new Error(body.message || "文件上传失败"));
      } catch { reject(new Error("上传响应无法解析，请刷新列表确认结果")); }
    };
    xhr.send(form);
  });
}
```

### 接口层：校验 + 可选转发 FastAPI

```ts
// src/app/api/v1/file/uploadFile/route.ts
export async function POST(request: Request) {
  try {
    if (Number(request.headers.get("content-length")) > MAX_FILE_BYTES + 1024 * 1024)
      return apiFail(413, "文件大小超过 50MB 限制");
    const form = await request.formData();
    const file = form.get("file");
    if (!file || typeof file === "string" || !file.name) return apiFail(400, "未提供文件");
    const type = file.name.split(".").pop()?.toLowerCase() || "";
    if (!isFileType(type)) return apiFail(415, `不支持的文件类型：${type || "未知"}`);
    if (!file.size) return apiFail(400, "文件内容为空");
    if (file.size > MAX_FILE_BYTES) return apiFail(413, "文件大小超过 50MB 限制");

    const input: UploadInput = {
      originalName: file.name, bytes: Buffer.from(await file.arrayBuffer()),
      collectionName: String(form.get("collection_name") || ""),
      description: String(form.get("description") || ""),
      chunkStrategy: String(form.get("chunk_strategy") || "adaptive"),
      createdBy: Number(form.get("created_by") || 1),
      storedName: buildStoredName(file.name),
    };
    normalizeUpload(input, MAX_FILE_BYTES);

    const forwarded = await forwardDocument(input);         // ← FastAPI 在线则走它
    if (forwarded) return apiOk(forwarded.message, forwarded.file, null, 201);
    return apiOk("文件上传成功", await saveUploadedFile(input), null, 201);
  } catch (error) { return fromError(error); }
}
```

```ts
  // 原 Python 白名单不含图片，图片始终走 Next.js 适配层

  const cfg = getBackendConfig();

    if (!response.ok) return null;
```

```ts
// src/lib/object-storage.ts
export function buildStoredName(original: string) {
  const leaf = original.split(/[/\\]/).pop() || "file";
  const safe = leaf.replace(/[^\w\-.\u4e00-\u9fa5]/g, "_");   // 保留中文
  const dot = safe.lastIndexOf(".");
  const extension = dot > 0 ? safe.slice(dot).toLowerCase() : "";
  const base = (dot > 0 ? safe.slice(0, dot) : safe) || "file";
  return `${base.slice(0, 180)}_${randomUUID().replace(/-/g, "").slice(0, 8)}${extension}`;
}
```

```ts
export async function uploadToMinioBucket(name, bytes, contentType, bucket = getBackendConfig().kbBucketName, originalName?) {
  await ensureMinioBucket(bucket);
  await withTimeout(getMinioClient().putObject(bucket, name, bytes, bytes.length, {
    "Content-Type": contentType,
    ...(originalName ? { "X-Amz-Meta-Original-Name": encodeURIComponent(originalName) } : {}),

  cacheObjectMetadata(bucket, name, { size: bytes.length, originalName, contentType });

  return { storedName: name, bucket, objectPath: `${bucket}/${name}` };
```

### 服务端校验（`records.ts`）

```ts
export function normalizeUpload(input: UploadInput, maxBytes: number) {
  if (!input.originalName || input.originalName.length > 255) throw new ApiError(400, "文件名须为 1–255 个字符");
  const fileType = input.originalName.split(".").pop()?.toLowerCase() || "";
  if (!isFileType(fileType)) throw new ApiError(415, `不支持的文件类型：${fileType || "未知"}`);
  if (!input.bytes.length) throw new ApiError(400, "文件内容为空");
  if (input.bytes.length > maxBytes) throw new ApiError(413, "文件大小超过 50MB 限制");
  validateImage(input.bytes, fileType);                       // ← 图片查魔数，防伪造扩展名
  const collectionName = input.collectionName.trim();
  if (!collectionName || collectionName.length > 256) throw new ApiError(400, "集合名称须为 1–256 个字符");
  const description = input.description?.trim() || null;
  if (description && description.length > 255) throw new ApiError(400, "描述不能超过 255 个字符");
  const chunkStrategy = (input.chunkStrategy || DEFAULT_CHUNK_STRATEGY).trim();
  if (!isChunkStrategy(chunkStrategy)) throw new ApiError(400, "不支持的分块策略");
  const createdBy = input.createdBy ?? 1;
  if (!Number.isSafeInteger(createdBy) || createdBy < 1 || createdBy > 2147483647) throw new ApiError(400, "创建者 ID 不合法");
  return { fileType, contentType: CONTENT_TYPES[fileType], collectionName, description, chunkStrategy, createdBy };
}
```

图片不信任扩展名，按文件头魔数判定：

```ts
// src/lib/file-content.ts
export function imageContentType(bytes: Uint8Array): string | null {
  const at = (offset: number, signature: number[]) => signature.every((value, i) => bytes[offset + i] === value);
  if (at(0, [137, 80, 78, 71, 13, 10, 26, 10])) return "image/png";
  if (at(0, [255, 216, 255])) return "image/jpeg";
  if (at(0, [71, 73, 70, 56]) && [55, 57].includes(bytes[4]) && bytes[5] === 97) return "image/gif";
  if (at(0, [82, 73, 70, 70]) && at(8, [87, 69, 66, 80])) return "image/webp";
  return null;
}
```

```ts
export function searchTokens(value: string) {
  return value.normalize("NFKC").trim().toLocaleLowerCase("zh-CN").split(/\s+/u).filter(Boolean).slice(0, 12);
}
export function matchesSearch(query: string, fields: (string | null | undefined)[]) {
  const values = fields.map((value) => (value || "").normalize("NFKC").toLocaleLowerCase("zh-CN"));
  return searchTokens(query).every((token) => values.some((value) => value.includes(token)));  // AND 语义
```

### MySQL 搜索 SQL

用 `LOCATE` 而非 `LIKE`，避免用户输入的 `%` `_` 被当成通配符：

```ts
// src/lib/mysql-search.ts
/** MySQL 5.7+ 兼容；把后端生成的 _<8位十六进制> 后缀还原成显示名 */
export const displayFileNameSql = sql`CASE
  WHEN file_name REGEXP ${"_[[:xdigit:]]{8}[.][^.]+$"}
  THEN CONCAT(LEFT(file_name, CHAR_LENGTH(file_name) - CHAR_LENGTH(SUBSTRING_INDEX(file_name, '.', -1)) - 10), '.', SUBSTRING_INDEX(file_name, '.', -1))
  ELSE file_name END`;

export function mysqlFileWhere(query: ListQuery): SQL {
  const normalized = normalizeQuery(query);
  const clauses: SQL[] = [];
  if (normalized.collectionName) clauses.push(sql`collection_name = ${normalized.collectionName}`);
  if (normalized.status) clauses.push(sql`status = ${normalized.status}`);
  if (normalized.fileType) clauses.push(sql`LOWER(file_type) = ${normalized.fileType}`);
  for (const token of searchTokens(normalized.q || "")) {
    // LOCATE 把 %、_、引号、反斜杠都当普通字符
    clauses.push(sql`(
      LOCATE(${token}, LOWER(COALESCE(file_name, ''))) > 0 OR
      LOCATE(${token}, LOWER(COALESCE(${displayFileNameSql}, ''))) > 0 OR
      LOCATE(${token}, LOWER(COALESCE(collection_name, ''))) > 0 OR
      LOCATE(${token}, LOWER(COALESCE(description, ''))) > 0 OR
      LOCATE(${token}, LOWER(COALESCE(file_type, ''))) > 0
    )`);
  }
  return clauses.length ? sql`WHERE ${sql.join(clauses, sql` AND `)}` : sql``;
}
```

并发度限制在 6，避免一次 `statObject` 风暴：

```ts
async function mapLimit<T, U>(items: T[], task: (item: T) => Promise<U>): Promise<U[]> {
  const result = new Array<U>(items.length);
  let index = 0;
  await Promise.all(Array.from({ length: Math.min(items.length, 6) }, async () => {
    while (index < items.length) { const current = index++; result[current] = await task(items[current]); }
  }));
  return result;
}
```

```tsx
// src/components/file-drawer.tsx
type Draft = { collection_name: string; description: string; status: string; chunk_strategy: string; updated_by: number };

const draftOf = (file: FileDict): Draft => ({
  collection_name: file.collection_name, description: file.description, status: file.status,
  chunk_strategy: file.chunk_strategy || "adaptive",
  updated_by: file.updated_by || file.created_by || 1,
});

// 五个字段任一变化都算「有修改」，操作人单独改也能保存
const dirty = Boolean(file && draft && (
  draft.collection_name !== file.collection_name ||
  draft.description !== file.description ||
  draft.status !== file.status ||
  draft.chunk_strategy !== file.chunk_strategy ||
  draft.updated_by !== (file.updated_by || file.created_by || 1)
));

// 有未保存修改时离开要确认
const leave = (next: () => void) => {
  if (!saving && !replacing && (!dirty || window.confirm("还有未保存的修改，确定离开文件信息吗？"))) next();
};

async function save() {
  if (!file || !draft) return;
  setSaving(true); setFeedback("");
  try {
    const next = await updateFile(file.id, draft);
    setFile(next); setDraft(draftOf(next)); onChanged(next);
    setFeedback(`已保存，当前版本 v${next.version_no}`);
    onToast("ok", "文件信息已保存");
  } catch (caught) { setError(caught instanceof Error ? caught.message : "保存失败"); }
  finally { setSaving(false); }
}
```

```ts
// src/lib/office-preview.ts
function tableForSheet(sheet: XLSX.WorkSheet): SpreadsheetTable {
  const ref = sheet["!ref"];
  if (!ref) return EMPTY_TABLE;
  const range = XLSX.utils.decode_range(ref);

  const hiddenRows = new Set<number>();
  (sheet["!rows"] || []).forEach((row, index) => { if (row?.hidden) hiddenRows.add(index); });
  const hiddenCols = new Set<number>();
  (sheet["!cols"] || []).forEach((col, index) => { if (col?.hidden) hiddenCols.add(index); });

  const rowIndexes: number[] = [];
  for (let r = range.s.r; r <= range.e.r; r += 1) if (!hiddenRows.has(r)) rowIndexes.push(r);
  const colIndexes: number[] = [];
  for (let c = range.s.c; c <= range.e.c; c += 1) if (!hiddenCols.has(c)) colIndexes.push(c);

  const shownRows = rowIndexes.slice(0, MAX_ROWS);     // 1000 行上限
  const shownCols = colIndexes.slice(0, MAX_COLUMNS);  // 80 列上限
  const rows = shownRows.map((r) => shownCols.map((c) => formatCell(sheet[XLSX.utils.encode_cell({ r, c })])));

  // 合并区裁剪到可见范围
  const rowPosition = new Map(shownRows.map((r, i) => [r, i]));
  const colPosition = new Map(shownCols.map((c, i) => [c, i]));
  const merges: SheetMerge[] = [];
  for (const merge of sheet["!merges"] || []) {
    const row = rowPosition.get(merge.s.r), col = colPosition.get(merge.s.c);
    if (row === undefined || col === undefined) continue;
    const rowSpan = shownRows.filter((r) => r >= merge.s.r && r <= merge.e.r).length;
    const colSpan = shownCols.filter((c) => c >= merge.s.c && c <= merge.e.c).length;
    if (rowSpan > 1 || colSpan > 1) merges.push({ row, col, row_span: rowSpan, col_span: colSpan });
  }
  return {
    rows,
    row_numbers: shownRows.map((r) => r + 1),                      // ← Excel 真实行号
    columns: shownCols.map((c) => XLSX.utils.encode_col(c)),       // ← Excel 真实列字母
    merges, total_rows: rowIndexes.length, total_columns: colIndexes.length,
    truncated: rowIndexes.length > MAX_ROWS || colIndexes.length > MAX_COLUMNS,
  };
}
```

```ts
  return process.env.KB_OPERATORS_FILE || path.join(process.cwd(), "storage", "operators.json");
}
const DEFAULTS: Operator[] = OPERATORS.map((operator) => ({ ...operator }));   // 1 林知衡 / 2 周简

export async function listOperators(): Promise<Operator[]> {
  try {
    const parsed = JSON.parse(await readFile(operatorsFile(), "utf8")) as { operators?: unknown };
    return sanitize(parsed.operators).length ? sanitize(parsed.operators) : [...DEFAULTS];
  } catch (error) {
    if ((error as NodeJS.ErrnoException).code === "ENOENT") return [...DEFAULTS];   // 文件不存在用默认
    if (error instanceof ApiError) throw error;
    throw new ApiError(503, "操作人名单读取失败，请检查 storage/operators.json");
  }
}

export async function addOperator(input: { name: unknown; id?: unknown }): Promise<Operator> {
  return serialize(async () => {                 // ← 串行化，避免并发分配到同一个 ID
    const name = validName(input.name);
    const list = await listOperators();
    if (list.some((operator) => operator.name === name)) throw new ApiError(409, "该操作人姓名已存在");
    let id: number;
    if (input.id != null && input.id !== "") {
      id = Number(input.id);
      if (!Number.isSafeInteger(id) || id < 1 || id > 2147483647) throw new ApiError(400, "操作人 ID 须为正整数");
      if (list.some((operator) => operator.id === id)) throw new ApiError(409, `操作人 ID ${id} 已被占用`);
    } else {
      id = list.reduce((max, operator) => Math.max(max, operator.id), 0) + 1;      // 取最大 +1
    }
    const operator: Operator = { id, name };
    await writeOperators([...list, operator].sort((a, b) => a.id - b.id));
    return operator;
  });
}

export async function renameOperator(id: number, name: unknown): Promise<Operator> {
  return serialize(async () => {
    const nextName = validName(name);
    const list = await listOperators();
    const target = list.find((operator) => operator.id === id);
    if (!target) throw new ApiError(404, "操作人不存在");
    if (list.some((operator) => operator.id !== id && operator.name === nextName)) throw new ApiError(409, "该操作人姓名已存在");
    target.name = nextName;                      // ← ID 不动，历史记录自动跟随新名字
    await writeOperators(list);
    return target;
  });
}
```

## 模块 9：文档分片（透传）

**这是本项目唯一一个"前端完全不碰业务逻辑"的模块**：分词、切分、元数据、向量化、
写 Milvus 全部在你的 FastAPI 里完成，Next.js 只负责发请求和渲染结果，**也没有本地兜底实现**。

### 调用链

```

  → client-api.parseFileChunks({file_id, minio_path, file_name})
    → POST /api/v1/chunks/parseFile        ← Next.js 代理（无业务逻辑）
```

### 为什么是代理而不是重新实现

| 能力 | 归属 | 原因 |
| --- | --- | --- |
| Jieba 中文分词 | FastAPI | Python 生态，Node 无等价实现 |
| 800/100 切分参数 | FastAPI | `chunk_router.py` 里写死，页面不提供修改入口 |
| 元数据生成 | FastAPI | `MinIOMetadataStrategy` 决定字段 |

```ts
const PARSE_TIMEOUT_MS = 10 * 60 * 1000;

export async function forwardParseFile(payload: ParsePayload): Promise<{ message: string; data: unknown }> {
  const cfg = getBackendConfig();
  const url = `${cfg.fastapiUrl}/api/v1/chunks/parseFile`;

  let response: Response;
  try {
    response = await fetch(url, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(payload),            // ← 原样转发，不加工
      signal: AbortSignal.timeout(PARSE_TIMEOUT_MS),
      cache: "no-store",
    });
  } catch (error) {
    const aborted = error instanceof Error && (error.name === "TimeoutError" || error.name === "AbortError");
    if (aborted) {
      throw new ApiError(504, `解析超时（已等待 ${PARSE_TIMEOUT_MS / 60000} 分钟）。任务可能仍在后端运行，请查看后端日志确认结果，不要立即重复提交。`);
    }
    throw new ApiError(503, `无法连接解析服务 ${cfg.fastapiUrl}。请确认 FastAPI 已启动（backend/ 目录执行 python main.py）。`);
  }
  // …错误翻译见下
}
```

**关键处理：上游 500 的真实含义。** 你的 `chunk_router.parse_file` 捕获异常后只记日志，
既不 `raise` 也不 `return`，函数返回 `None`；FastAPI 用 `response_model=ApiResponse`
校验 `None` 会失败，最终回 500。所以 500 并不代表前端出错，而是后端处理失败且原因只在日志里：

```ts
if (response.status === 500) {
  throw new ApiError(502, "解析服务返回 500：请求已到达后端但未返回结果。常见原因是 Ollama、Milvus 或 MinIO 未就绪，详细原因请查看 FastAPI 控制台日志。");
}
```

还要防住"假成功"——上游回 200 但 `data` 为空时不能当作完成：

```ts
if (!body || body.data == null) {
  throw new ApiError(502, "解析服务未返回结果数据，请查看 FastAPI 控制台日志确认处理是否成功。");
}
```

## 附：对象路径解析（兼容多种历史格式）

`minio_path` 在不同来源下格式不同，统一由 `parseObjectKey` 解析：

```ts
// src/lib/object-storage.ts
/** minio_path 是权威来源：它可能指向另一个桶或多级 key */
export function parseObjectKey(minioPath?: string | null, fallbackName?: string | null) {
  let bucket = bucketName();
  let objectName = "";
  const raw = minioPath?.trim();
  if (raw && /^https?:\/\//i.test(raw)) {              // 旧版存的预签名 URL
    try {
      const url = new URL(raw);
      const parts = url.pathname.replace(/^\//, "").split("/");
      if (parts.length >= 2) { bucket = decodeURIComponent(parts.shift()!); objectName = decodeURIComponent(parts.join("/")); }
    } catch { throw new ApiError(400, "存储记录中的对象 URL 无法解析"); }
  } else if (raw?.includes("/")) {                     // bucket/key 形式
    const normalized = raw.replace(/^\/+/, "");
    const slash = normalized.indexOf("/");
    bucket = normalized.slice(0, slash);
    objectName = normalized.slice(slash + 1);
  }
  if (!objectName) objectName = fallbackName || raw || "";
  assertSafeBucket(bucket);
  assertSafeObjectName(objectName);                    // ← 阻断路径穿越
  return { bucket, objectName };
}
```

```ts
export function assertSafeObjectName(name: string) {
  if (!name || name.length > 1024 || name.startsWith("/") || /[\\\x00-\x1f]/.test(name) ||
      name.split("/").some((part) => !part || part === "." || part === "..")) {
    throw new ApiError(400, "对象名不合法");
  }
}
```

```ts
  const record = await findRow(id);
  const located = parseObjectKey(record.minio_path, record.file_name);
  let bytes: Buffer;
  try {
    bytes = await readFromMinioBucket(located.objectName, located.bucket);
  } catch (error) {
    try { bytes = await readObject(located.objectName, located.bucket); }   // 本地镜像兜底
    catch {
      const code = (error as { code?: string }).code;
      if (code === "NoSuchKey" || code === "NoSuchBucket" || code === "NotFound") {
        throw new ApiError(404, "记录存在，但 MinIO 中未找到对应文件。请检查存储桶和对象路径，或重新上传原文件。");
      }
      throw new ApiError(503, "文件暂时无法读取，请检查 MinIO 服务和访问权限后重试");
    }
  }
  return { row: { ...toDict(mapRow(record)), file_size: bytes.length }, bytes };
}
```
