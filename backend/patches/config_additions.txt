#
#  现状：ChatBotQueryRouter.__init__ 里执行 self.chunk_service = ChunkService()，
#  而 main.py 在模块级就构造了 ChatBotQueryRouter()。ChunkService 构造时会去连
#  Ollama 与 Milvus 并做嵌入测试 —— 只要这两个服务没起，整个 FastAPI 就起不来，
#  连原来的 /api/v1/file/uploadFile 也会不可用。
#
#  把 chatbot_query_router.py 里这两行：
#      self.chunk_service = ChunkService()  # 复用分片服务
#  改成：
#      self._chunk_service = None  # 复用分片服务（惰性初始化，避免拖垮启动）
#
#  再在类里加一个属性：
#      @property
#      def chunk_service(self):
#          if self._chunk_service is None:
#              self._chunk_service = ChunkService()
#          return self._chunk_service
#
#  这样只有真正用到分片时才初始化，FastAPI 启动不再依赖 Ollama/Milvus。

#  已知问题：chatbot_query_router.py 的异常分支会抛第二个异常
#
#  except (NotFoundException, ValidationException) as e:
#      return ApiResponse(status=e.code, ...)
#
#  NotFoundException / ValidationException 继承自 BaseAPIException(HTTPException)，
#  它们有 .status_code，但没有 .code（.code 是 RAGException 的属性）。
#  所以这一行会抛 AttributeError，把原本的 400/404 变成一个空泛的 500。
#
#  建议改为：
#      return ApiResponse(status=getattr(e, "status_code", 400), message=str(e.detail), data={})
